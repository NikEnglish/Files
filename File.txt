import chess
import chess.engine
import os

# Словарь для перевода фигур на русский
PIECES_RU = {
    chess.PAWN: "ПЕШКУ",
    chess.KNIGHT: "КОНЯ (N)",
    chess.BISHOP: "СЛОНА (B)",
    chess.ROOK: "ЛАДЬЮ (R)",
    chess.QUEEN: "ФЕРЗЯ (Q)",
    chess.KING: "КОРОЛЯ (K)"
}

def start_assistant():
    # Путь к экзешнику в той же папке
    engine_path = os.path.join(os.getcwd(), "stockfish.exe")

    if not os.path.exists(engine_path):
        print(f"!!! ОШИБКА: Положи stockfish.exe в папку: {os.getcwd()}")
        return

    try:
        engine = chess.engine.SimpleEngine.popen_uci(engine_path)
        # Уровень сложности как у Luk.AI (2500 ELO)
        engine.configure({"UCI_Elo": 3190, "UCI_LimitStrength": True})
        print("\n=== ПОМОЩНИК ПРОТИВ LUK.AI ЗАПУЩЕН ===")
    except Exception as e:
        print(f"Ошибка движка: {e}")
        return

    board = chess.Board()
    my_color = input("Твой цвет в игре (w - белые / b - черные): ").lower().strip()

    while not board.is_game_over():
        print("\n" + "-"*40)
        print(board)
        print("-"*40)
        
        is_my_turn = (board.turn == chess.WHITE and my_color == 'w') or \
                     (board.turn == chess.BLACK and my_color == 'b')

        if is_my_turn:
            print("\n>>> ДУМАЮ...")
            result = engine.play(board, chess.engine.Limit(time=1.5))
            move = result.move
            
            # Получаем инфо о ходе
            piece_type = board.piece_at(move.from_square).piece_type
            piece_name = PIECES_RU.get(piece_type)
            target_square = chess.square_name(move.to_square)
            san_move = board.san(move) # Красивый вид типа Ne7
            
            print(f"\n[ ХОДИ ТАК ]: Бери {piece_name} и ставь на {target_square}")
            print(f"[ ЗАПИСЬ ХОДА ]: {san_move}")
            
            board.push(move)
            input("\nНажми Enter, когда сделаешь ход в телефоне...")
        else:
            print("\n<<< ЖДУ ХОД ПРОТИВНИКА")
            move_input = input("Как сходил Luk.AI? (например, e4 или Bc3): ").strip()
            
            try:
                # Понимает и твои Bc3, и координаты e2e4
                try:
                    move = board.parse_san(move_input)
                except:
                    move = board.parse_uci(move_input)
                
                board.push(move)
            except:
                print("!! Ошибка: Не понял ход. Пиши как в Chess.com (например, Nf3 или d5)")

    print("\nПАРТИЯ ОКОНЧЕНА!")
    print(f"Результат: {board.result()}")
    engine.quit()

if __name__ == "__main__":
    start_assistant()
