# -------------------- DATABASE FUNCTIONS --------------------

def init_db():
    """Initializes the database with proper error handling"""
    conn = None
    try:
        conn = sqlite3.connect('bot_data.db')
        cur = conn.cursor()
        
        # Users table with all required columns
        cur.execute('''CREATE TABLE IF NOT EXISTS users (
            user_id INTEGER PRIMARY KEY,
            username TEXT,
            first_name TEXT,
            last_name TEXT,
            group_name TEXT,
            role TEXT DEFAULT 'student',
            custom_role_name TEXT,
            can_send_homework BOOLEAN DEFAULT 0,
            can_send_notifications BOOLEAN DEFAULT 0,
            can_send_global_notifications BOOLEAN DEFAULT 0,
            can_delete_homework BOOLEAN DEFAULT 0,
            can_delete_notifications BOOLEAN DEFAULT 0,
            can_delete_suggestions BOOLEAN DEFAULT 0,
            can_manage_users BOOLEAN DEFAULT 0,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )''')
        
        # Group roles table
        cur.execute('''CREATE TABLE IF NOT EXISTS group_roles (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            group_name TEXT NOT NULL,
            role_type TEXT NOT NULL,
            user_id INTEGER,
            username TEXT,
            first_name TEXT,
            assigned_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            UNIQUE(group_name, role_type),
            FOREIGN KEY (user_id) REFERENCES users (user_id) ON DELETE SET NULL
        )''')
        
        # User notification settings table
        cur.execute('''CREATE TABLE IF NOT EXISTS user_notification_settings (
            user_id INTEGER PRIMARY KEY,
            lesson_start BOOLEAN DEFAULT 1,
            new_homework BOOLEAN DEFAULT 1,
            important_notifications BOOLEAN DEFAULT 1,
            tomorrow_schedule BOOLEAN DEFAULT 1,
            today_schedule BOOLEAN DEFAULT 1,
            FOREIGN KEY (user_id) REFERENCES users (user_id) ON DELETE CASCADE
        )''')
        
        # Homeworks table
        cur.execute('''CREATE TABLE IF NOT EXISTS homeworks (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            subject TEXT,
            task TEXT,
            deadline TEXT,
            group_name TEXT,
            added_by INTEGER,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            FOREIGN KEY (added_by) REFERENCES users (user_id)
        )''')
        
        # Homework media table
        cur.execute('''CREATE TABLE IF NOT EXISTS homework_media (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            homework_id INTEGER,
            media_type TEXT,
            media_id TEXT,
            caption TEXT,
            FOREIGN KEY (homework_id) REFERENCES homeworks (id) ON DELETE CASCADE
        )''')
        
        # Notifications table
        cur.execute('''CREATE TABLE IF NOT EXISTS notifications (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            message TEXT,
            group_name TEXT,
            added_by INTEGER,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            FOREIGN KEY (added_by) REFERENCES users (user_id)
        )''')
        
        # Notification media table
        cur.execute('''CREATE TABLE IF NOT EXISTS notification_media (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            notification_id INTEGER,
            media_type TEXT,
            media_id TEXT,
            caption TEXT,
            FOREIGN KEY (notification_id) REFERENCES notifications (id) ON DELETE CASCADE
        )''')
        
        # Suggestions table
        cur.execute('''CREATE TABLE IF NOT EXISTS suggestions (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id INTEGER,
            message TEXT,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            FOREIGN KEY (user_id) REFERENCES users (user_id)
        )''')
        
        # Suggestion media table
        cur.execute('''CREATE TABLE IF NOT EXISTS suggestion_media (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            suggestion_id INTEGER,
            media_type TEXT,
            media_id TEXT,
            caption TEXT,
            FOREIGN KEY (suggestion_id) REFERENCES suggestions (id) ON DELETE CASCADE
        )''')
        
        # Suggestion responses table
        cur.execute('''CREATE TABLE IF NOT EXISTS suggestion_responses (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            suggestion_id INTEGER,
            responder_id INTEGER,
            message TEXT,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            FOREIGN KEY (suggestion_id) REFERENCES suggestions (id),
            FOREIGN KEY (responder_id) REFERENCES users (user_id)
        )''')
        
        # Birthdays table
        cur.execute('''CREATE TABLE IF NOT EXISTS birthdays (
            user_id INTEGER PRIMARY KEY,
            birthday_date TEXT,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            FOREIGN KEY (user_id) REFERENCES users (user_id) ON DELETE CASCADE
        )''')
        
        # Global notifications table
        cur.execute('''CREATE TABLE IF NOT EXISTS global_notifications (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            message TEXT,
            added_by INTEGER,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            FOREIGN KEY (added_by) REFERENCES users (user_id)
        )''')
        
        # Global notification media table
        cur.execute('''CREATE TABLE IF NOT EXISTS global_notification_media (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            global_notification_id INTEGER,
            media_type TEXT,
            media_id TEXT,
            caption TEXT,
            FOREIGN KEY (global_notification_id) REFERENCES global_notifications (id) ON DELETE CASCADE
        )''')
        
        conn.commit()
        log_event("База данных инициализирована.")
    except sqlite3.Error as e:
        log_error(f"Ошибка базы данных при инициализации: {e}")
    finally:
        if conn:
            conn.close()

def get_db_connection():
    """Creates a connection to the database"""
    try:
        conn = sqlite3.connect('bot_data.db')
        conn.row_factory = sqlite3.Row
        return conn
    except sqlite3.Error as e:
        log_error(f"Ошибка базы данных при подключении: {e}")
        return None

def get_user(user_id):
    """Gets user information"""
    conn = get_db_connection()
    if not conn: return None
    try:
        user = conn.execute('SELECT * FROM users WHERE user_id = ?', (user_id,)).fetchone()
        return user
    except sqlite3.Error as e:
        log_error(f"Ошибка базы данных при получении пользователя {user_id}: {e}")
        return None
    finally:
        conn.close()

def get_group_role(group_name, role_type):
    """Gets role for specific group"""
    conn = get_db_connection()
    if not conn: return None
    try:
        role = conn.execute(
            'SELECT * FROM group_roles WHERE group_name = ? AND role_type = ?', 
            (group_name, role_type)
        ).fetchone()
        return role
    except sqlite3.Error as e:
        log_error(f"Ошибка базы данных при получении роли группы: {e}")
        return None
    finally:
        conn.close()

def set_group_role(group_name, role_type, user_id, username, first_name):
    """Sets role for specific group"""
    conn = get_db_connection()
    if not conn: return False
    try:
        conn.execute('''INSERT OR REPLACE INTO group_roles 
                     (group_name, role_type, user_id, username, first_name) 
                     VALUES (?, ?, ?, ?, ?)''',
                     (group_name, role_type, user_id, username, first_name))
        
        rights_data = {}
        if role_type == ROLE_STAROSTA:
            rights_data = {
                'can_send_homework': 1,
                'can_send_notifications': 1,
                'can_delete_homework': 1,
                'can_delete_notifications': 1,
                'role': ROLE_STAROSTA
            }
        elif role_type == ROLE_DEPUTY_STAROSTA:
            rights_data = {
                'can_send_homework': 1,
                'can_send_notifications': 1,
                'can_delete_homework': 1,
                'can_delete_notifications': 1,
                'role': ROLE_DEPUTY_STAROSTA
            }
        
        if rights_data:
            conn.execute('''UPDATE users SET
                         can_send_homework = ?,
                         can_send_notifications = ?,
                         can_delete_homework = ?,
                         can_delete_notifications = ?,
                         role = ?
                         WHERE user_id = ?''',
                         (rights_data.get('can_send_homework', 0),
                          rights_data.get('can_send_notifications', 0),
                          rights_data.get('can_delete_homework', 0),
                          rights_data.get('can_delete_notifications', 0),
                          rights_data.get('role', ROLE_STUDENT),
                          user_id))
        
        conn.commit()
        log_event(f"Установлена роль '{role_type}' для группы '{group_name}' — {first_name} (ID: {user_id})")
        return True
    except sqlite3.Error as e:
        log_error(f"Ошибка базы данных при установке роли группы: {e}")
        return False
    finally:
        conn.close()

def remove_group_role(group_name, role_type):
    """Removes role from group"""
    conn = get_db_connection()
    if not conn: return False
    try:
        role = get_group_role(group_name, role_type)
        if role and role['user_id']:
            conn.execute('''UPDATE users SET
                         can_send_homework = 0,
                         can_send_notifications = 0,
                         can_delete_homework = 0,
                         can_delete_notifications = 0,
                         role = 'student'
                         WHERE user_id = ?''', (role['user_id'],))
        
        conn.execute('DELETE FROM group_roles WHERE group_name = ? AND role_type = ?', 
                    (group_name, role_type))
        conn.commit()
        log_event(f"Удалена роль '{role_type}' для группы '{group_name}'")
        return True
    except sqlite3.Error as e:
        log_error(f"Ошибка базы данных при удалении роли группы: {e}")
        return False
    finally:
        conn.close()

# -------------------- SPAM PROTECTION --------------------

def check_spam(user_id):
    now = datetime.now().timestamp()
    if user_id in user_blocks and now < user_blocks[user_id]:
        return True, user_blocks[user_id] - now
    
    if user_id not in user_message_times:
        user_message_times[user_id] = []
        user_warnings[user_id] = 0
    
    user_message_times[user_id].append(now)
    user_message_times[user_id] = [t for t in user_message_times[user_id] if now - t < SPAM_WINDOW]
    
    if len(user_message_times[user_id]) >= SPAM_THRESHOLD:
        user_warnings[user_id] += 1
        warning_count = user_warnings[user_id]
        if warning_count - 1 < len(BLOCK_DURATIONS):
            block_duration = BLOCK_DURATIONS[warning_count - 1]
        else:
            block_duration = BLOCK_DURATIONS[-1]
        user_blocks[user_id] = now + block_duration
        user_message_times[user_id] = []
        log_event(f"Обнаружен спам от пользователя {user_id}! Блокировка на {block_duration} сек.", level="ВНИМАНИЕ")
        return True, block_duration
    return False, 0

def fetch_groups_list():
    global ALL_GROUPS, groups_cache, last_groups_check
    try:
        schedule_data = fetch_schedule_data()
        if not schedule_data:
            return ALL_GROUPS
        
        groups = set()
        for lesson in schedule_data:
            if 'group' in lesson:
                groups.add(lesson['group'])
        
        def group_sort_key(group_name):
            try:
                parts = group_name.split('-')
                if len(parts) > 1:
                    number_part = parts[1]
                    for char in number_part:
                        if char.isdigit():
                            course = int(char)
                            return (course, group_name)
            except:
                pass
            return (0, group_name)
        
        sorted_groups = sorted(groups, key=group_sort_key)
        ALL_GROUPS = sorted_groups
        groups_cache = sorted_groups
        last_groups_check = datetime.now(YEKATERINBURG_TZ)
        log_event(f"Обновлен список групп. Всего групп: {len(ALL_GROUPS)}")

def parse_lesson_data(lesson):
    try:
        if not isinstance(lesson, dict):
            return None
        lesson_data = {}
        lesson_data['group'] = lesson.get('group', 'Не указана')
        lesson_data['number'] = lesson.get('number', 0)
        lesson_data['type'] = lesson.get('type', 'Занятие')
        lesson_data['subject'] = lesson.get('subject', 'Не указано')
        lesson_data['campus'] = lesson.get('campus', 'Не указан')
        lesson_data['room'] = lesson.get('room', 'Не указана')
        lesson_data['teacher'] = lesson.get('teacher', 'Не указан')
        
        start_time_str = lesson.get('startTime', '')
        end_time_str = lesson.get('endTime', '')
        
        if start_time_str:
            try:
                start_time_utc = datetime.fromisoformat(start_time_str.replace('Z', '+00:00'))
                start_time_yekat = start_time_utc.astimezone(YEKATERINBURG_TZ)
                lesson_data['date'] = start_time_yekat.strftime('%Y-%m-%d')
                lesson_data['start_time'] = start_time_yekat.strftime('%H:%M')
                lesson_data['datetime'] = start_time_yekat
            except:
                lesson_data['date'] = 'Не указана'
                lesson_data['start_time'] = 'Не указано'
        
        if end_time_str:
            try:
                end_time_utc = datetime.fromisoformat(end_time_str.replace('Z', '+00:00'))
                end_time_yekat = end_time_utc.astimezone(YEKATERINBURG_TZ)
                lesson_data['end_time'] = end_time_yekat.strftime('%H:%M')
            except:
                lesson_data['end_time'] = 'Не указано'
        
        return lesson_data
    except Exception as e:
        log_error(f"Не удалось распарсить данные занятия: {e}")
        return None

def get_type_emoji(lesson_type):
    emoji_map = {
        'Занятие': '📚', 'Практика': '🔬', 'Лекция': '📖', 'Экзамен': '📝',
        'Консультация': '💬', 'Лабораторная работа': '⚗️', 'Семинар': '💡', 'Зачет': '✅'
    }
    return emoji_map.get(lesson_type, '📚')

def find_merged_groups(current_lesson, all_parsed_lessons):
    merged = []
    current_group = current_lesson['group']
    for lesson in all_parsed_lessons:
        if (lesson['group'] != current_group and 
            lesson['datetime'].date() == current_lesson['datetime'].date() and 
            lesson['start_time'] == current_lesson['start_time'] and 
            lesson['end_time'] == current_lesson['end_time'] and 
            lesson['room'] == current_lesson['room'] and 
            lesson['campus'] == current_lesson['campus'] and 
            lesson['teacher'] == current_lesson['teacher'] and 
            lesson['subject'] == current_lesson['subject'] and 
            lesson['number'] == current_lesson['number']):
            merged.append(lesson['group'])
    return merged

def format_schedule(lessons, title, all_parsed_lessons=None):
    try:
        if not lessons:
            return f"✅ Нет занятий для {title.split('(')[-1].split(')')[0]}"
        lessons.sort(key=lambda x: (x.get('datetime', datetime.min), x.get('number', 0)))
        result = [f"<b>{title}</b>\n"]
        current_date = None
        for lesson in lessons:
            try:
                if 'datetime' in lesson:
                    date_str = lesson['datetime'].strftime('%d.%m.%Y')
                    day_names = ['Понедельник', 'Вторник', 'Среда', 'Четверг', 'Пятница', 'Суббота', 'Воскресенье']
                    day_of_week = day_names[lesson['datetime'].weekday()]
                    if date_str != current_date:
                        current_date = date_str
                        result.append(f"\n📅 <b>{date_str} ({day_of_week})</b>")
                    type_emoji = get_type_emoji(lesson['type'])
                    start_time = lesson.get('start_time', '❓')
                    end_time = lesson.get('end_time', '❓')
                    lesson_text = (
                        f"    {type_emoji} {lesson['type']}\n"
                        f"    ⏰ {start_time}-{end_time} | {lesson['number']} пара\n"
                        f"    📚 {lesson['subject']}\n"
                        f"    👨‍🏫 {lesson['teacher']}\n"
                        f"    🚪 {lesson['room']}({lesson['campus']})\n"
                    )
                    if all_parsed_lessons:
                        merged = find_merged_groups(lesson, all_parsed_lessons)
                        if merged:
                            lesson_text += f"    👥 Совмещена с группой {', '.join(merged)}\n"
                    result.append(lesson_text)
            except Exception as e:
                log_error(f"Не удалось отформатировать пару: {e}")
                continue
        result.append(f"\n📊 <b>Всего занятий:</b> {len(lessons)}")
        return "\n".join(result)
    except Exception as e:
        return f"❌ Ошибка при формировании расписания: {str(e)}"

def get_schedule(period, group_name):
    try:
        global schedule_cache, last_schedule_check
        now = datetime.now(YEKATERINBURG_TZ)
        if last_schedule_check is None or (now - last_schedule_check).total_seconds() > 3600:
            log_event(f"Обновляем кэш расписания для группы {group_name}")
            schedule_data = fetch_schedule_data()
            if schedule_data:
                schedule_cache = schedule_data
                last_schedule_check = now
                log_event("Расписание обновлено из API")
            else:
                log_event("Не удалось обновить расписание, используем кэш если есть")
        else:
            schedule_data = schedule_cache
            log_event(f"Используем кэшированное расписание для группы {group_name}")
        
        if not schedule_data:
            return "❌ Не удалось загрузить расписание. Попробуйте позже."
        
        group_lessons = []
        for lesson in schedule_data:
            if lesson.get('group') == group_name:
                parsed = parse_lesson_data(lesson)
                if parsed and 'datetime' in parsed:
                    group_lessons.append(parsed)
        
        if not group_lessons:
            return f"❌ Нет расписания для группы {group_name}"
        
        all_parsed_lessons = []
        for lesson in schedule_data:
            parsed = parse_lesson_data(lesson)
            if parsed and 'datetime' in parsed:
                all_parsed_lessons.append(parsed)
        
        today_yekat = datetime.now(YEKATERINBURG_TZ).date()
        
        if period == '📅 Сегодня':
            filtered_lessons = [l for l in group_lessons if l['datetime'].date() == today_yekat]
            title = f"📅 Расписание на сегодня ({today_yekat.strftime('%d.%m.%Y')}) - {group_name}"
            return format_schedule(filtered_lessons, title, all_parsed_lessons)
            
        elif period == '📅 Завтра':
            tomorrow_yekat = today_yekat + timedelta(days=1)
            filtered_lessons = [l for l in group_lessons if l['datetime'].date() == tomorrow_yekat]
            title = f"📅 Расписание на завтра ({tomorrow_yekat.strftime('%d.%m.%Y')}) - {group_name}"
            return format_schedule(filtered_lessons, title, all_parsed_lessons)
            
        elif period == '📅 Неделя':
            start_of_week = today_yekat - timedelta(days=today_yekat.weekday())
            end_of_week = start_of_week + timedelta(days=6)
            if today_yekat.weekday() >= 5:
                start_of_week += timedelta(days=7)
                end_of_week += timedelta(days=7)
            filtered_lessons = [l for l in group_lessons if start_of_week <= l['datetime'].date() <= end_of_week]
            title = f"📅 Расписание на неделю ({start_of_week.strftime('%d.%m')}-{end_of_week.strftime('%d.%m')}) - {group_name}"
            return format_schedule(filtered_lessons, title, all_parsed_lessons)
        else:
            return None, split_group_schedule_by_days(group_lessons, group_name, all_parsed_lessons)
    except Exception as e:
        log_error(f"Произошла ошибка при получении расписания: {str(e)}")
        return f"❌ Произошла ошибка при загрузке расписания: {str(e)}"

def split_group_schedule_by_days(lessons, group_name, all_parsed_lessons):
    try:
        if not lessons: return []
        lessons.sort(key=lambda x: x['datetime'])
        days = {}
        for lesson in lessons:
            date_key = lesson['datetime'].date()
            if date_key not in days:
                days[date_key] = []
            days[date_key].append(lesson)
        
        messages = []
        for date in sorted(days.keys()):
            day_lessons = days[date]
            date_str = date.strftime('%d.%m.%Y')
            day_name = ['Понедельник', 'Вторник', 'Среда', 'Четверг', 'Пятница', 'Суббота', 'Воскресенье'][date.weekday()]
            title = f"📅 <b>{date_str} ({day_name})</b> — Группа {group_name}"
            formatted = format_schedule(day_lessons, title, all_parsed_lessons)
            messages.append(formatted)
        return messages
    except Exception as e:
        log_error(f"Ошибка при разбиении расписания группы по дням: {e}")
        return []

async def check_schedule_changes(context: ContextTypes.DEFAULT_TYPE):
    global schedule_cache, last_schedule_check, teachers_cache
    log_event("Плановая проверка изменений в расписании...")
    new_schedule = fetch_schedule_data()
    if not new_schedule: return

    if schedule_cache:
        old_parsed = [parse_lesson_data(l) for l in schedule_cache if parse_lesson_data(l) and 'datetime' in parse_lesson_data(l)]
        new_parsed = [parse_lesson_data(l) for l in new_schedule if parse_lesson_data(l) and 'datetime' in parse_lesson_data(l)]
        
        old_dict = {(l['group'], l['datetime'].date(), l['start_time']): l for l in old_parsed}
        new_dict = {(l['group'], l['datetime'].date(), l['start_time']): l for l in new_parsed}
        
        changes = []
        for key, new_lesson in new_dict.items():
            old_lesson = old_dict.get(key)
            if old_lesson:
                if (old_lesson['subject'] != new_lesson['subject'] or
                    old_lesson['teacher'] != new_lesson['teacher'] or
                    old_lesson['room'] != new_lesson['room'] or
                    old_lesson['campus'] != new_lesson['campus']):
                    changes.append(('changed', old_lesson, new_lesson))
            else:
                changes.append(('added', new_lesson))
        
        for key, old_lesson in old_dict.items():
            if key not in new_dict:
                changes.append(('removed', old_lesson))
        
        if changes:
            log_event(f"Обнаружено {len(changes)} изменений в расписании!")
            for change in changes:
                if change[0] == 'changed':
                    _, old, new = change
                    group = new['group']
                    old_text = (
                        f"    📚 {old['type']}\n"
                        f"    ⏰ {old['start_time']}-{old['end_time']} | {old['number']} пара\n"
                        f"    📚 {old['subject']}\n"
                        f"    👨‍🏫 {old['teacher']}\n"
                        f"    🚪 {old['room']}({old['campus']})"
                    )
                    new_text = (
                        f"    📚 {new['type']}\n"
                        f"    ⏰ {new['start_time']}-{new['end_time']} | {new['number']} пара\n"
                        f"    📚 {new['subject']}\n"
                        f"    👨‍🏫 {new['teacher']}\n"
                        f"    🚪 {new['room']}({new['campus']})"
                    )
                    message = f"🔄 Изменение в расписании (замена):\n\n{old_text}\n\nна:\n\n{new_text}"
                    conn = get_db_connection()
                    users = conn.execute('SELECT user_id FROM users WHERE group_name = ?', (group,)).fetchall()
                    conn.close()
                    for user in users:
                        try:
                            await context.bot.send_message(user['user_id'], message, parse_mode='HTML')
                        except Exception as e:
                            log_error(f"Не удалось отправить уведомление об изменении: {e}")
    
    schedule_cache = new_schedule
    last_schedule_check = datetime.now(YEKATERINBURG_TZ)
    fetch_teachers_list()

def format_call_schedule(schedule, day_type):
    """
    Форматирует расписание звонков:
    - 5 минут: выводится как есть (внутри пары)
    - 10 минут: просто отступ (пустая строка между парами)
    - большая перемена: выводится блоком (не трогается)
    """
    result = [f"🔔 <b>РАСПИСАНИЕ ЗВОНКОВ</b>\n📅 <i>{day_type}</i>\n"]
    
    if isinstance(schedule, list):
        for item in schedule:
            if item['type'] == 'lesson':
                if item['part'] == 1:
                    result.append(f"📚 <b>{item['number']} ПАРА</b>")
                    result.append(f"  ├ 🟢 <b>{item['start']} - {item['end']}</b> <i>(1-я часть)</i>")
                else:
                    result.append(f"  └ 🟢 <b>{item['start']} - {item['end']}</b> <i>(2-я часть)</i>")
            else:  # break
                try:
                    duration = int(item.get('duration', 0))
                except (ValueError, TypeError):
                    duration = 0
                
                if duration == 5:
                    result.append(f"  ├ ☕ {item['start']} - {item['end']} <i>({duration} мин)</i>")
                elif duration == 10:
                    result.append("")
                else:
                    duration_str = f"({duration} мин)" if duration else ""
                    result.append(f"\n🛌 <b>Большая перемена</b>\n  └ ⏰ <b>{item['start']} - {item['end']}</b> <i>{duration_str}</i>\n")
    else:
        for key, value in schedule.items():
            if isinstance(key, int):
                result.append(f"📚 <b>{key} ПАРА</b>\n  └ 🟢 <b>{value['start']} - {value['end']}</b>")
            else:
                try:
                    duration = int(value.get('duration', 0))
                except (ValueError, TypeError):
                    duration = 0
                
                if duration == 5:
                    result.append(f"  ├ ☕ {value['start']} - {value['end']} <i>({duration} мин)</i>")
                elif duration == 10:
                    result.append("")
                else:
                    duration_str = f"({duration} мин)" if duration else ""
                    result.append(f"\n🛌 <b>Большая перемена:</b> ⏰ <b>{value['start']} - {value['end']}</b> <i>{duration_str}</i>\n")
    
    formatted_text = "\n".join(result)
    while "\n\n\n" in formatted_text:
        formatted_text = formatted_text.replace("\n\n\n", "\n\n")
    return formatted_text

async def check_lesson_notifications(context: ContextTypes.DEFAULT_TYPE):
    try:
        global schedule_cache
        if not schedule_cache: return
        now = datetime.now(YEKATERINBURG_TZ)
        today = now.date()
        
        groups_lessons = {}
        for lesson in schedule_cache:
            parsed = parse_lesson_data(lesson)
            if parsed and 'datetime' in parsed and parsed['datetime'].date() == today:
                group_name = parsed['group']
                if group_name not in groups_lessons:
                    groups_lessons[group_name] = []
                groups_lessons[group_name].append(parsed)
        
        for group_name, lessons in groups_lessons.items():
            for lesson in lessons:
                lesson_time = lesson['datetime']
                time_diff = (lesson_time - now).total_seconds()
                if 240 <= time_diff <= 300:
                    log_event(f"Уведомление за 5 минут до пары: {group_name} — {lesson['subject']}")
                    await send_lesson_notification(context.bot, group_name, lesson)
    except Exception as e:
        log_error(f"Ошибка в check_lesson_notifications: {e}")

async def handle_birthday_input(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_id = update.effective_user.id
    text = update.message.text.strip()
    log_event(f"Пользователь {user_id} ввел дату рождения: '{text}'")
    if text == '❌ Отмена':
        return await cancel_conversation(update, context)
    
    try:
        if not re.match(r'^\d{1,2}\.\d{1,2}$', text):
            raise ValueError("Неверный формат")
        day, month = map(int, text.split('.'))
        if month < 1 or month > 12 or day < 1 or day > 31:
            raise ValueError("Некорректная дата")
        birthday_date = f"{day:02d}.{month:02d}"
        set_user_birthday(user_id, birthday_date)
        await update.message.reply_text(
            f"✅ Дата рождения установлена: {birthday_date}\n"
            f"Теперь бот будет поздравлять вас каждый год!",
            reply_markup=ReplyKeyboardRemove()
        )
    except Exception:
        log_event(f"Пользователь {user_id} указал некорректный формат даты рождения: '{text}'")
        await update.message.reply_text("❌ Неверный формат! Введите ДД.ММ (например 15.09):", parse_mode='HTML')
        return AWAITING_BIRTHDAY
    
    await show_main_menu(update, user_id)
    return ConversationHandler.END

        return await cancel_conversation(update, context)
    
    text = update.message.text if update.message.text else (update.message.caption if update.message.caption else "")

        return await cancel_conversation(update, context)

        notif_id = add_global_notification(notification_text, user_id)
        if notif_id:
            for m in media_collection:
                add_global_notification_media(notif_id, m['type'], m['id'], m.get('caption'))

            clear_media_collection(user_id, 'global_notification')
            context.user_data.pop('global_notification_text', None)
        await show_main_menu(update, user_id)
        return ConversationHandler.END

    if update.message.photo:
        add_to_media_collection(user_id, 'global_notification', {'type': 'photo', 'id': update.message.photo[-1].file_id, 'caption': update.message.caption})
        await update.message.reply_text("✅ Фото добавлено. Отправьте еще медиа или '✅ Готово'.")
        return AWAITING_GLOBAL_NOTIFICATION
    elif update.message.video:

        await update.message.reply_text("✅ Видео добавлено. Отправьте еще медиа или '✅ Готово'.")

    elif update.message.document:

        await update.message.reply_text("✅ Текст сохранен. Отправьте медиа или '✅ Готово'.")

async def send_global_notification_to_all(bot, global_notification_id, message, added_by):
    role_display = get_user_role_display(added_by)
    global_text = f"🌍 <b>ГЛОБАЛЬНОЕ УВЕДОМЛЕНИЕ</b>\n\n{message}\n\n👤 <b>От:</b> {role_display}"
    conn = get_db_connection()
    media = conn.execute('SELECT * FROM global_notification_media WHERE global_notification_id = ?', (global_notification_id,)).fetchall()
    conn.close()
    
    users = get_all_users()
    for user in users:
        try:
            settings = get_user_notification_settings(user['user_id'])
            if settings and settings['important_notifications']:
                if media:
                    if len(media) > 1:
                        media_group = []
                        for i, m in enumerate(media):
                            m_dict = dict(m)
                            cap = global_text if i == 0 else None
                            if m_dict['media_type'] == 'photo':
                                media_group.append(InputMediaPhoto(m_dict['media_id'], caption=cap, parse_mode='HTML'))
                            elif m_dict['media_type'] == 'video':
                                media_group.append(InputMediaVideo(m_dict['media_id'], caption=cap, parse_mode='HTML'))
                            elif m_dict['media_type'] == 'document':
                                media_group.append(InputMediaDocument(m_dict['media_id'], caption=cap, parse_mode='HTML'))
                        for i in range(0, len(media_group), 10):
                            await bot.send_media_group(chat_id=user['user_id'], media=media_group[i:i+10])
                    else:
                        m_dict = dict(media[0])
                        if m_dict['media_type'] == 'photo':
                            await bot.send_photo(user['user_id'], photo=m_dict['media_id'], caption=global_text, parse_mode='HTML')
                        elif m_dict['media_type'] == 'video':
                            await bot.send_video(user['user_id'], video=m_dict['media_id'], caption=global_text, parse_mode='HTML')
                        elif m_dict['media_type'] == 'document':
                            await bot.send_document(user['user_id'], document=m_dict['media_id'], caption=global_text, parse_mode='HTML')
                else:
                    await bot.send_message(user['user_id'], global_text, parse_mode='HTML')

        await show_main_menu(update, update.effective_user.id)
        return ConversationHandler.END

async def handle_homework_input(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_id = update.effective_user.id
    user_data = get_user(user_id)
    if not can_send_homework(user_id):
        return await cancel_conversation(update, context)

    group_name = user_data['group_name']
    text = update.message.text if update.message.text else (update.message.caption if update.message.caption else "")
    if text and text.strip() == '❌ Отмена':
        log_event(f"Пользователь {user_id} отменил отправку ДЗ")
        return await cancel_conversation(update, context)
    
    if text and text.strip() == '✅ Готово':
        media_collection = get_media_collection(user_id, 'homework')
        hw_text = context.user_data.get('homework_text', '')
        if not hw_text and not media_collection:
            await update.message.reply_text("❌ Пожалуйста, добавь описание к ДЗ или медиафайлы.")
            return AWAITING_HOMEWORK
        if not hw_text:
            await update.message.reply_text("❌ Пожалуйста, добавь описание к ДЗ.")
            return AWAITING_HOMEWORK
        lines = hw_text.strip().split('\n')
        if len(lines) >= 3:
            hw_id = add_homework(lines[0].strip(), lines[1].strip(), lines[2].strip(), group_name, user_id)
            if hw_id:
                for m in media_collection:
                    add_homework_media(hw_id, m['type'], m['id'], m.get('caption'))
                log_event(f"Пользователь {user_id} опубликовал ДЗ #{hw_id} для группы {group_name}")
                await update.message.reply_text("✅ ДЗ добавлено и отправляется...", reply_markup=ReplyKeyboardRemove())
                await send_homework_to_all(context.bot, hw_id, lines[0].strip(), lines[1].strip(), lines[2].strip(), group_name, user_id)
                clear_media_collection(user_id, 'homework')
                context.user_data.pop('homework_text', None)
            await show_main_menu(update, user_id)
            return ConversationHandler.END
        else:
            await update.message.reply_text("❌ Неправильный формат. Нужно 3 строки: предмет, задание, срок")
            return AWAITING_HOMEWORK

    if update.message.photo:
        add_to_media_collection(user_id, 'homework', {'type': 'photo', 'id': update.message.photo[-1].file_id, 'caption': update.message.caption})
        await update.message.reply_text("✅ Фото добавлено. Отправьте еще медиа или нажмите '✅ Готово'.")
        return AWAITING_HOMEWORK
    elif update.message.video:
        add_to_media_collection(user_id, 'homework', {'type': 'video', 'id': update.message.video.file_id, 'caption': update.message.caption})
        await update.message.reply_text("✅ Видео добавлено. Отправьте еще медиа или нажмите '✅ Готово'.")
        return AWAITING_HOMEWORK
    elif update.message.document:
        add_to_media_collection(user_id, 'homework', {'type': 'document', 'id': update.message.document.file_id, 'caption': update.message.caption})
        await update.message.reply_text("✅ Документ добавлен. Отправьте еще медиа или нажмите '✅ Готово'.")
        return AWAITING_HOMEWORK

    if text and text.strip():
        context.user_data['homework_text'] = text
        log_event(f"Пользователь {user_id} сохранил текст ДЗ")
        await update.message.reply_text("✅ Текст ДЗ сохранен. Отправьте медиа или нажмите '✅ Готово'.")
        return AWAITING_HOMEWORK
    return AWAITING_HOMEWORK

async def send_homework_to_all(bot, homework_id, subject, task, deadline, group_name, added_by):
    role_display = get_user_role_display(added_by)
    hw_text = (
        f"📝 <b>НОВОЕ ДОМАШНЕЕ ЗАДАНИЕ</b>\n\n"
        f"📚 <b>Предмет:</b> {subject}\n"
        f"📋 <b>Задание:</b> {task}\n"
        f"⏰ <b>Срок сдачи:</b> {deadline}\n"
        f"👥 <b>Группа:</b> {group_name}\n"
        f"👤 <b>От:</b> {role_display}"
    )
    conn = get_db_connection()
    media = conn.execute('SELECT * FROM homework_media WHERE homework_id = ?', (homework_id,)).fetchall()
    conn.close()
    users = get_db_connection().execute('SELECT user_id FROM users WHERE group_name = ?', (group_name,)).fetchall()
    log_event(f"Рассылка ДЗ #{homework_id} для {len(users)} пользователей группы {group_name}")
    for user in users:
        try:
            settings = get_user_notification_settings(user['user_id'])
            if settings and settings['new_homework']:
                if media:
                    if len(media) > 1:
                        media_group = []
                        for i, m in enumerate(media):
                            m_dict = dict(m)
                            cap = hw_text if i == 0 else None
                            if m_dict['media_type'] == 'photo':
                                media_group.append(InputMediaPhoto(m_dict['media_id'], caption=cap, parse_mode='HTML'))
                            elif m_dict['media_type'] == 'video':
                                media_group.append(InputMediaVideo(m_dict['media_id'], caption=cap, parse_mode='HTML'))
                            elif m_dict['media_type'] == 'document':
                                media_group.append(InputMediaDocument(m_dict['media_id'], caption=cap, parse_mode='HTML'))
                        for i in range(0, len(media_group), 10):
                            await bot.send_media_group(chat_id=user['user_id'], media=media_group[i:i+10])
                    else:
                        m_dict = dict(media[0])
                        if m_dict['media_type'] == 'photo':
                            await bot.send_photo(chat_id=user['user_id'], photo=m_dict['media_id'], caption=hw_text, parse_mode='HTML')
                        elif m_dict['media_type'] == 'video':
                            await bot.send_video(chat_id=user['user_id'], video=m_dict['media_id'], caption=hw_text, parse_mode='HTML')
                        elif m_dict['media_type'] == 'document':
                            await bot.send_document(chat_id=user['user_id'], document=m_dict['media_id'], caption=hw_text, parse_mode='HTML')
                else:
                    await bot.send_message(user['user_id'], hw_text, parse_mode='HTML')
        except Exception as e:
            log_error(f"Ошибка при доставке ДЗ пользователю {user['user_id']}: {e}")

async def show_users_list(update: Update):
    user_id = update.effective_user.id
    log_event(f"Пользователь {user_id} запросил список всех зарегистрированных пользователей")
    users = get_all_users()
    if not users:
        await update.message.reply_text("❌ В базе данных нет пользователей.")
        return
    groups_dict = {}
    for user in users:
        group_name = user['group_name'] if user['group_name'] else "Без группы"
        if group_name not in groups_dict:
            groups_dict[group_name] = []
        groups_dict[group_name].append(user)
    
    user_list_text = "👥 <b>Список всех пользователей по группам</b>\n\n"
    for group_name, group_users in sorted(groups_dict.items()):
        user_list_text += f"<b>📚 Группа: {group_name}</b> ({len(group_users)} чел.)\n"
        for user in group_users:
            username = user['username'] if user['username'] else "Нет юзернейма"
            role_display = get_user_role_display(user['user_id'])
            user_list_text += (
                f"  👤 <b>Имя:</b> {user['first_name']}\n"
                f"  🆔 <b>ID:</b> <code>{user['user_id']}</code>\n"
                f"  👨‍💼 <b>Роль:</b> {role_display}\n"
                f"  📱 <b>Юзер:</b> @{username}\n\n"
            )
        user_list_text += "\n"
    
    if len(user_list_text) > 4000:
        parts, current_part = [], ""
        for section in user_list_text.split('\n\n'):
            if len(current_part + section + '\n\n') > 4000:
                parts.append(current_part)
                current_part = section + '\n\n'
            else:
                current_part += section + '\n\n'
        if current_part: parts.append(current_part)
        for i, part in enumerate(parts):
            await update.message.reply_text(f"{part}\n<b>Часть {i+1}/{len(parts)}</b>", parse_mode='HTML')
    else:
        await update.message.reply_text(user_list_text, parse_mode='HTML')

async def handle_role_type_selection(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()
    if query.data == "role_cancel": return await cancel_conversation(update, context)
    if query.data.startswith("role_back_to_users_"):
        group_name = query.data.replace('role_back_to_users_', '')
        context.user_data['role_assignment_group'] = group_name
        return await handle_role_group_selection(update, context)
    if query.data == "custom_role":
        log_event("Разработчик запросил ввод кастомного имени роли")
        await query.edit_message_text("✏️ <b>Назначение собственной роли</b>\n\nВведите название роли для пользователя:", parse_mode='HTML', reply_markup=ReplyKeyboardMarkup([['❌ Отмена']], resize_keyboard=True))
        return AWAITING_CUSTOM_ROLE

    user_id = context.user_data['role_assignment_user']
    group_name = context.user_data['role_assignment_group']
    
    if query.data == "role_remove_all":
        log_event(f"Разработчик снял все роли в группе {group_name}")
        remove_group_role(group_name, ROLE_STAROSTA)
        remove_group_role(group_name, ROLE_DEPUTY_STAROSTA)
        await query.edit_message_text("✅ Все роли сняты!")
    else:
        role_type = query.data.replace('role_type_', '')
        user_data = get_user(user_id)
        log_event(f"Разработчик назначил роль '{role_type}' для {user_data['first_name']} в группе {group_name}")
        set_group_role(group_name, role_type, user_id, user_data['username'], user_data['first_name'])
        await query.edit_message_text("✅ Роль успешно назначена!")
    await show_main_menu(update, query.from_user.id)
    return ConversationHandler.END

async def handle_custom_role_input(update: Update, context: ContextTypes.DEFAULT_TYPE):
    try:
        user_id = update.effective_user.id
        if user_id != DEVELOPER_ID: return await cancel_conversation(update, context)
        custom_role_name = update.message.text.strip()
        if custom_role_name == '❌ Отмена': return await start_role_assignment(update, context)
        
        if context.user_data.get('custom_role_global'):
            try:
                target_user_id = int(custom_role_name)
                context.user_data['custom_role_user_id'] = target_user_id
                context.user_data.pop('custom_role_global', None)
                log_event(f"Разработчик ввел целевой ID для кастомной роли: {target_user_id}")
                await update.message.reply_text("✏️ <b>Назначение собственной роли</b>\n\nВведите название роли для пользователя:", parse_mode='HTML', reply_markup=ReplyKeyboardMarkup([['❌ Отмена']], resize_keyboard=True))
                return AWAITING_CUSTOM_ROLE
            except ValueError:
                await update.message.reply_text("❌ Неверный формат ID пользователя. Введите числовой ID:", reply_markup=ReplyKeyboardMarkup([['❌ Отмена']], resize_keyboard=True))
                return AWAITING_CUSTOM_ROLE
        else:
            target_user_id = context.user_data.get('custom_role_user_id', context.user_data.get('role_assignment_user'))
            target_user = get_user(target_user_id)
            if not target_user:
                await update.message.reply_text("❌ Пользователь не найден.")
                return ConversationHandler.END
            
            rights_data = dict(target_user)
            rights_data['role'] = ROLE_CUSTOM
            rights_data['custom_role_name'] = custom_role_name
            update_user_rights(target_user_id, rights_data)
            log_event(f"Разработчик установил кастомную роль '{custom_role_name}' пользователю {target_user['first_name']} (ID: {target_user_id})")
            await update.message.reply_text(f"✅ Пользователю {target_user['first_name']} установлена роль: {custom_role_name}!", reply_markup=ReplyKeyboardRemove())
            await show_main_menu(update, user_id)
            return ConversationHandler.END
    except Exception as e:
        log_error(f"Ошибка в handle_custom_role_input: {e}")
        return AWAITING_CUSTOM_ROLE

# -------------------- USER RIGHTS MANAGEMENT --------------------

async def manage_user_rights_menu(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_id = update.effective_user.id
    log_event(f"Пользователь {user_id} открыл управление правами пользователей")
    if user_id != DEVELOPER_ID and not can_manage_users(user_id): return ConversationHandler.END
    users = get_all_users()
    groups_dict = {}
    for user in users:
        if user['user_id'] == update.effective_user.id: continue
        gname = user['group_name'] or "Без группы"
        groups_dict.setdefault(gname, []).append(user)
    keyboard = [[InlineKeyboardButton(f"📚 {g} ({len(groups_dict[g])} чел.)", callback_data=f"rights_group_{g}")] for g in sorted(groups_dict.keys())]
    await update.message.reply_text("👥 <b>Управление правами пользователей</b>\n\nВыберите группу:", parse_mode='HTML', reply_markup=InlineKeyboardMarkup(keyboard))
    return AWAITING_USER_RIGHTS

async def handle_user_rights_group_selection(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()
    if query.data == "rights_back_to_groups": return await manage_user_rights_menu(update, context)
    group_name = query.data.replace('rights_group_', '')
    context.user_data['rights_management_group'] = group_name
    log_event(f"Выбрана группа {group_name} для управления правами")
    users = [u for u in get_all_users() if u['group_name'] == group_name and u['user_id'] != query.from_user.id]
    keyboard = [[InlineKeyboardButton(f"{u['first_name']} (@{u['username']}) - {get_user_role_display(u['user_id'])}", callback_data=f"manage_rights_{u['user_id']}")] for u in users]
    keyboard.append([InlineKeyboardButton("🔙 Назад к группам", callback_data="rights_back_to_groups")])
    await query.edit_message_text(f"👥 <b>Управление правами группы {group_name}</b>\n\nВыберите пользователя:", parse_mode='HTML', reply_markup=InlineKeyboardMarkup(keyboard))
    return AWAITING_USER_RIGHTS

async def handle_rights_toggle(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()
    if query.data == "save_rights":
        log_event(f"Пользователь {query.from_user.id} сохранил права пользователей")
        await query.edit_message_text("✅ Права сохранены!")
        await show_main_menu(update, query.from_user.id)
        return ConversationHandler.END
    if query.data.startswith("rights_back_to_users_"):
        group_name = query.data.replace('rights_back_to_users_', '')
        context.user_data['rights_management_group'] = group_name
        return await handle_user_rights_group_selection(update, context)

    parts = query.data.split('_')
    if len(parts) == 3:
        action = parts[1]
        target_user_id = int(parts[2])
    elif len(parts) == 4:
        action = f"{parts[1]}_{parts[2]}"
        target_user_id = int(parts[3])
    else:
        return

    target_user = get_user(target_user_id)
    rights_data = dict(target_user)
    log_event(f"Переключение права '{action}' для пользователя {target_user_id}")
    if action == "reset":
        for key in list(rights_data.keys()):
            if key.startswith('can_'): rights_data[key] = 0
        rights_data['role'] = ROLE_STUDENT
        rights_data['custom_role_name'] = None
    elif action == "homework": rights_data['can_send_homework'] = 1 - rights_data['can_send_homework']
    elif action == "notifications": rights_data['can_send_notifications'] = 1 - rights_data['can_send_notifications']
    elif action == "global": rights_data['can_send_global_notifications'] = 1 - rights_data['can_send_global_notifications']
    elif action == "del_homework": rights_data['can_delete_homework'] = 1 - rights_data['can_delete_homework']
    elif action == "del_notifications": rights_data['can_delete_notifications'] = 1 - rights_data['can_delete_notifications']
    elif action == "del_suggestions": rights_data['can_delete_suggestions'] = 1 - rights_data['can_delete_suggestions']
    elif action == "manage_users": rights_data['can_manage_users'] = 1 - rights_data['can_manage_users']

    update_user_rights(target_user_id, rights_data)
    updated_user = get_user(target_user_id)
    keyboard = [
        [InlineKeyboardButton(f"{'✅' if updated_user['can_send_homework'] else '❌'} Отправка ДЗ", callback_data=f"toggle_homework_{target_user_id}")],
        [InlineKeyboardButton(f"{'✅' if updated_user['can_send_notifications'] else '❌'} Отправка уведомлений", callback_data=f"toggle_notifications_{target_user_id}")],
        [InlineKeyboardButton(f"{'✅' if updated_user['can_send_global_notifications'] else '❌'} Глобальные уведомления", callback_data=f"toggle_global_{target_user_id}")],
        [InlineKeyboardButton(f"{'✅' if updated_user['can_delete_homework'] else '❌'} Удаление ДЗ", callback_data=f"toggle_del_homework_{target_user_id}")],
        [InlineKeyboardButton(f"{'✅' if updated_user['can_delete_notifications'] else '❌'} Удаление уведомлений", callback_data=f"toggle_del_notifications_{target_user_id}")],
        [InlineKeyboardButton(f"{'✅' if updated_user['can_delete_suggestions'] else '❌'} Удаление предложений", callback_data=f"toggle_del_suggestions_{target_user_id}")],
        [InlineKeyboardButton(f"{'✅' if updated_user['can_manage_users'] else '❌'} Управление пользователями", callback_data=f"toggle_manage_users_{target_user_id}")],
        [InlineKeyboardButton("🔙 Назад к списку", callback_data=f"rights_back_to_users_{context.user_data['rights_management_group']}")],
        [InlineKeyboardButton("❌ Сбросить все права", callback_data=f"reset_rights_{target_user_id}")],
        [InlineKeyboardButton("✅ Сохранить и выйти", callback_data="save_rights")]
    ]
    role_display = get_user_role_display(target_user_id)
    await query.edit_message_text(f"👤 <b>Управление правами: {updated_user['first_name']}</b>\nГруппа: {updated_user['group_name']} | Роль: {role_display}\n\nНажмите на право для изменения:", parse_mode='HTML', reply_markup=InlineKeyboardMarkup(keyboard))
    return AWAITING_USER_RIGHTS
